function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var data = {}; if (e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (err) { data = e.parameter || {}; } } else { data = e.parameter || {}; } sheet.appendRow([ new Date().toLocaleString("he-IL"), data.name || data.by || "אורח", data.status || data.type || "", Number(data.guests) || 0, data.ride || "", data.title || "", data.dj || "", data.score || "" ]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; // תמיכה גם בשמירת נתונים דרך פרמטרים בכתובת (למקרה ש-POST נחסם) if (e && e.parameter && e.parameter.action === "save") { var p = e.parameter; sheet.appendRow([ new Date().toLocaleString("he-IL"), p.name || p.by || "אורח", p.status || p.type || "", Number(p.guests) || 0, p.ride || "", p.title || "", p.dj || "", p.score || "" ]); return ContentService.createTextOutput("SAVED").setMimeType(ContentService.MimeType.TEXT); } var rows = sheet.getDataRange().getValues(); var list = []; for (var i = 0; i < rows.length; i++) { var r = rows[i]; if (r[2] === "מגיע" || r[2] === "אולי" || r[2] === "לא מגיע") { list.push({ date: String(r[0]), name: String(r[1]), status: String(r[2]), guests: Number(r[3]) || 1, ride: String(r[4] || "-") }); } } return ContentService .createTextOutput(JSON.stringify({ rsvps: list.reverse() })) .setMimeType(ContentService.MimeType.JSON); }
